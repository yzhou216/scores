%% Manuscript paper with a circle of fifths and a keyboard diagram.  The key
%% signatures are engraved by LilyPond (keysigs.ly); the rest is TikZ.
%%
%% Coordinates are in bp (PDF points) from the top-left corner of the page.

%% Computer Modern ships fixed design sizes; without this the accidentals
%% below 5pt get rounded up to the nearest available one.
\RequirePackage{fix-cm}

\documentclass{article}

\usepackage[paperwidth=612bp,paperheight=792bp,margin=0bp]{geometry}
\usepackage{graphicx}
\usepackage{tgtermes}
\usepackage{tikz}

%% Loaded last, as hyperref patches the internals of what comes before it.
%% hidelinks keeps the footer links free of the default coloured boxes.
\usepackage[hidelinks]{hyperref}

\pagestyle{empty}

%% Snippets from keysigs.ly, built from the repository root or in place.
\graphicspath{{build/manuscript/}{../../build/manuscript/}}

%% Staves

\newcommand{\StaffLeft}{30.6}
\newcommand{\StaffWidth}{550.44}
\newcommand{\StaffLineGap}{10.8}
\newcommand{\StaffRule}{0.48}
\newcommand{\StaffFirst}{201.06}   % centre of the topmost line
\newcommand{\StaffPitch}{71.1936}  % staff to staff
\newcommand{\StaffCount}{7}        % eight staves, zero-indexed

%% Keyboard

\newcommand{\KeyTop}{73.92}
\newcommand{\KeyBottom}{181.44}
\newcommand{\KeyFirst}{206.64}     % centre of the leftmost rule
\newcommand{\KeyWidth}{18.846}     % one white key
\newcommand{\KeyCount}{19}         % white keys, A up to E
\newcommand{\KeyRule}{0.96}
\newcommand{\BlackWidth}{14.52}
\newcommand{\BlackBottom}{124.32}
\newcommand{\BoxTop}{100.32}       % write-in box inset into each black key
\newcommand{\BoxBottom}{111.36}
\newcommand{\BoxInset}{0.96}

%% Gaps carrying a black key.  The keyboard runs A..E, so the ones without
%% are the B-C and E-F gaps: 2, 5, 9, 12 and 16.
\newcommand{\BlackKeys}{1,3,4,6,7,8,10,11,13,14,15,17,18}

%% Circle of fifths

\newcommand{\CircleX}{105.84}      % centre of the diagram
\newcommand{\CircleY}{118.32}
\newcommand{\RingInner}{33.52}
\newcommand{\RingOuter}{42.84}

%% Radii shared by all twelve sectors.  Angles run anticlockwise from three
%% o'clock, so C sits at 90.
\newcommand{\MinorRadius}{32}
\newcommand{\CountRadius}{40}
\newcommand{\MajorRadius}{50}
\newcommand{\StaveRadius}{63}
\newcommand{\StaveOuter}{81}       % second spelling of an enharmonic sector
\newcommand{\CaptionRadius}{60}    % "Major", above C
\newcommand{\CaptionInner}{25}     % "Minor", inside the ring
\newcommand{\KeysigScale}{0.46}

\definecolor{ringgray}{gray}{0.922}
\definecolor{footergray}{gray}{0.502}

%% Place #3 at #1 degrees, #2 bp out from the centre of the diagram.
\newcommand{\atangle}[3]{%
  \pgfmathsetmacro{\ptx}{\CircleX + (#2) * cos(#1)}%
  \pgfmathsetmacro{\pty}{\CircleY - (#2) * sin(#1)}%
  \node[inner sep=0pt] at (\ptx,\pty) {#3}%
}

\newcommand{\keysig}[1]{\includegraphics[scale=\KeysigScale]{keysigs-#1.cropped}}

\begin{document}
%% y points down, so every coordinate reads (from left, from top).
\begin{tikzpicture}[remember picture, overlay, x=1bp, y=-1bp,
                    shift={(current page.north west)}]

  \foreach \st in {0,...,\StaffCount}{
    \foreach \li in {0,...,4}{
      \pgfmathsetmacro{\y}{\StaffFirst + \st*\StaffPitch + \li*\StaffLineGap}
      \fill (\StaffLeft, \y-0.5*\StaffRule)
            rectangle +(\StaffWidth, \StaffRule);
    }
  }

  %% Rules run the full depth; the black keys and their boxes paint over them.
  \foreach \kk in {0,...,\KeyCount}{
    \pgfmathsetmacro{\x}{\KeyFirst + \kk*\KeyWidth}
    \fill (\x-0.5*\KeyRule, \KeyTop) rectangle (\x+0.5*\KeyRule, \KeyBottom);
  }
  \pgfmathsetmacro{\kbdright}{\KeyFirst + \KeyCount*\KeyWidth}
  \fill (\KeyFirst-0.5*\KeyRule, \KeyBottom-\KeyRule)
        rectangle (\kbdright+0.5*\KeyRule, \KeyBottom);
  \foreach \kk in \BlackKeys{
    \pgfmathsetmacro{\x}{\KeyFirst + \kk*\KeyWidth}
    \fill (\x-0.5*\BlackWidth, \KeyTop)
          rectangle (\x+0.5*\BlackWidth, \BlackBottom);
    \fill[white] (\x-0.5*\BlackWidth+\BoxInset, \BoxTop)
                 rectangle (\x+0.5*\BlackWidth-\BoxInset, \BoxBottom);
  }

  \fill[ringgray, even odd rule]
    (\CircleX,\CircleY) circle (\RingOuter bp)
    (\CircleX,\CircleY) circle (\RingInner bp);

  \atangle{90}{\CaptionRadius}{\fontsize{4.5bp}{6bp}\selectfont Major};
  \atangle{90}{\CaptionInner}{\fontsize{4.5bp}{6bp}\selectfont Minor};

  %% angle / major key / relative minor / accidental count
  \foreach \a/\maj/\rel/\cnt in {%
    90/C/a/{$\natural$},
    60/G/e/{1$\sharp$},
    30/D/b/{2$\sharp$},
    0/A/{f$\sharp$}/{3$\sharp$},
    -30/E/{c$\sharp$}/{4$\sharp$},
    -60/B/{g$\sharp$}/{7$\flat$/5$\sharp$},
    -90/{G$\flat$/F$\sharp$}/{e$\flat$/d$\sharp$}/{6$\flat$/6$\sharp$},
    -120/{D$\flat$}/{b$\flat$}/{5$\flat$/7$\sharp$},
    -150/{A$\flat$}/f/{4$\flat$},
    180/{E$\flat$}/c/{3$\flat$},
    150/{B$\flat$}/g/{2$\flat$},
    120/F/d/{1$\flat$}%
  }{
    \atangle{\a}{\MajorRadius}{\fontsize{8bp}{9bp}\selectfont\maj};
    \atangle{\a}{\CountRadius}{\fontsize{5.5bp}{7bp}\selectfont\cnt};
    \atangle{\a}{\MinorRadius}{\fontsize{7bp}{8bp}\selectfont\rel};
  }

  %% C major has no signature, so only eight sectors take a single one.
  \foreach \a/\name in {%
    60/g-major, 30/d-major, 0/a-major, -30/e-major,
    120/f-major, 150/b-flat-major, 180/e-flat-major, -150/a-flat-major%
  }{
    \atangle{\a}{\StaveRadius}{\keysig{\name}};
  }

  %% The rest show both spellings, nudged off their sector angle to stay
  %% clear of the key names.
  \foreach \a/\rad/\name in {%
    -52/\StaveRadius/b-major,       -52/\StaveOuter/c-flat-major,
    -128/\StaveRadius/d-flat-major, -128/\StaveOuter/c-sharp-major,
    -80/\StaveRadius/f-sharp-major, -100/\StaveRadius/g-flat-major%
  }{
    \atangle{\a}{\rad}{\keysig{\name}};
  }

  \node[anchor=base, footergray, inner sep=0pt] at (306,754.08)
    {\fontsize{7.955bp}{9bp}\selectfont
     \textcopyright{} 2025-2026
     \href{https://yiyuzhou.io/}{Yiyu Zhou} \ \textbullet\ %
     Licensed under \href{https://spdx.org/licenses/0BSD.html}{0BSD} OR
     \href{https://creativecommons.org/publicdomain/zero/1.0/legalcode}{CC0-1.0}};

\end{tikzpicture}
\end{document}
